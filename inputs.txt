const dadosVigaReta = {
  matriz_rigidez: [
    [12, -6, 0],
    [-6, 8, -4],
    [0, -4, 10]
  ],
  vetor_forcas: [10, 0, -5],
  deslocamentos_conhecidos: [0, 0, 0]
};

for (let i = 0; i < 3; i++) {
  for (let j = 0; j < 3; j++) {
    const rigidezInput = document.querySelector(`input[name="rigidez_r_${i}_${j}"]`);
    if (rigidezInput) rigidezInput.value = dadosVigaReta.matriz_rigidez[i][j];
  }
}

for (let i = 0; i < 3; i++) {
  const forcaInput = document.querySelector(`input[name="forcas_r_${i}"]`);
  const deslocInput = document.querySelector(`input[name="deslocamentos_r_${i}"]`);
  if (forcaInput) forcaInput.value = dadosVigaReta.vetor_forcas[i];
  if (deslocInput) deslocInput.value = dadosVigaReta.deslocamentos_conhecidos[i];
}

  dadosVigaPoligonal.distancias.join(',');
